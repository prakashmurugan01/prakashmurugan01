# 🌿 Plant Disease Detection — CNN Architecture Pipeline
import tensorflow as tf
import cv2
import numpy as np
from django.http import JsonResponse

class PlantDiseaseDetector:
    """
    CNN-powered agricultural AI achieving 95%+ detection accuracy
    Diseases covered: 40+ | Processing time: < 2 seconds
    """
    MODEL_PATH    = "models/plant_disease_cnn_v3.h5"
    IMG_SIZE      = (224, 224)
    CONFIDENCE_THRESHOLD = 0.85

    def __init__(self):
        self.model   = tf.keras.models.load_model(self.MODEL_PATH)
        self.classes = self._load_disease_classes()   # 40+ disease types

    def preprocess(self, img_path: str) -> np.ndarray:
        img = cv2.imread(img_path)
        img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
        img = cv2.resize(img, self.IMG_SIZE)
        img = cv2.GaussianBlur(img, (3, 3), 0)       # noise reduction
        return tf.keras.applications.resnet50.preprocess_input(img)

    def predict(self, img_path: str) -> dict:
        tensor    = np.expand_dims(self.preprocess(img_path), axis=0)
        probs     = self.model.predict(tensor)[0]
        top_idx   = np.argmax(probs)
        confidence = float(probs[top_idx])

        return {
            "disease"     : self.classes[top_idx],
            "confidence"  : f"{confidence * 100:.2f}%",
            "treatment"   : self._get_treatment(top_idx),
            "severity"    : self._calc_severity(confidence),
            "is_healthy"  : self.classes[top_idx] == "Healthy",
        }

# 🏫 Xplore IT Hub — AI-Powered LMS Core Modules
from django.db import models
from django.contrib.auth.models import AbstractUser
import cv2, numpy as np

# ── Role-based User Architecture ────────────────────────────────────────
class User(AbstractUser):
    ROLES = [("student","Student"),("teacher","Teacher"),("admin","Admin")]
    role  = models.CharField(max_length=10, choices=ROLES)

class Course(models.Model):
    title      = models.CharField(max_length=200)
    instructor = models.ForeignKey(User, on_delete=models.CASCADE)
    ai_summary = models.TextField(blank=True)   # auto-generated by NLP

# ── Face Recognition Attendance Engine ──────────────────────────────────
class AttendanceEngine:
    """OpenCV-powered automatic attendance via facial recognition"""

    def __init__(self):
        self.recognizer = cv2.face.LBPHFaceRecognizer_create()
        self.detector   = cv2.CascadeClassifier(
            cv2.data.haarcascades + 'haarcascade_frontalface_default.xml'
        )

    def mark_attendance(self, frame: np.ndarray) -> list[str]:
        gray    = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        faces   = self.detector.detectMultiScale(gray, 1.3, 5)
        present = []
        for (x, y, w, h) in faces:
            label, conf = self.recognizer.predict(gray[y:y+h, x:x+w])
            if conf < 70:                          # confidence threshold
                present.append(self._get_student(label))
        return present

# ── NLP Chatbot for Student Queries ─────────────────────────────────────
class AIChatbot:
    """Intelligent query processing with intent recognition"""
    INTENTS = {
        "schedule"    : r"(timetable|class|when|schedule)",
        "assignment"  : r"(assignment|submit|deadline|due)",
        "attendance"  : r"(attendance|present|absent|mark)",
    }
    def respond(self, query: str) -> str:
        import re
        for intent, pattern in self.INTENTS.items():
            if re.search(pattern, query.lower()):
                return self._fetch_answer(intent)
        return self._llm_fallback(query)
